get_filename_component(builtin "${CMAKE_CURRENT_BINARY_DIR}" NAME)
if(NOT builtin MATCHES "^(core_functions|parquet|icu)$")
    message(FATAL_ERROR "Unexpected prebuilt extension: ${builtin}")
endif()
set(archive "/build/prebuilt/lib${builtin}_extension.a")
if(NOT EXISTS "${archive}")
    message(FATAL_ERROR "Missing pinned extension archive: ${archive}")
endif()
add_library(${builtin}_extension STATIC IMPORTED GLOBAL)
set_target_properties(${builtin}_extension PROPERTIES IMPORTED_LOCATION "${archive}")
